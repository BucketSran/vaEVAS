from pathlib import Path
import sys,tempfile,unittest,subprocess,json
sys.path.insert(0,str(Path(__file__).parent));from operator_wrapper import supervise
class Controls(unittest.TestCase):
 def exercise(self,body,**kw):
  temp=tempfile.TemporaryDirectory(dir=Path(__file__).parent);self.addCleanup(temp.cleanup);root=Path(temp.name);case=root/'out/runs/test/base';case.mkdir(parents=True)
  return supervise([sys.executable,'-c',body,str(case)],root/'out',root/'receipt.json',interval=.02,**kw)
 def test_normal_and_nonzero_descendants_are_cleaned_before_reap(self):
  for code in (0,3):
   r=self.exercise('import subprocess,sys;subprocess.Popen([sys.executable,"-c","import time;time.sleep(30)"]);sys.exit('+str(code)+')')
   self.assertEqual(r['child_exit_code'],code);self.assertTrue(r['cleanup_confirmed']);self.assertTrue(any(e.get('action')=='kill_owned_unreaped_group' for e in r['events']));self.assertEqual(r['members_after_reap'],[])
 def test_file_limit_and_case_monitor_stop(self):
  r=self.exercise('import pathlib,sys;pathlib.Path(sys.argv[1],"big").write_bytes(b"x"*65536)',file_cap=4096)
  self.assertNotEqual(r['exit_code'],0);self.assertTrue(r['cleanup_confirmed'])
  r=self.exercise('import pathlib,sys,time;pathlib.Path(sys.argv[1],"big").write_bytes(b"x"*65536);time.sleep(30)',case_cap=4096)
  self.assertEqual(r['status'],'case_output_cap');self.assertEqual(r['members_after_reap'],[])
 def test_cancel_and_uncertain_cleanup_fail_closed(self):
  r=self.exercise('import time;time.sleep(30)',cancel_after=.05);self.assertEqual(r['status'],'cancelled');self.assertTrue(r['cleanup_confirmed'])
  def uncertain(*args):raise RuntimeError('uncertain cleanup negative control')
  r=self.exercise('pass',cleanup=uncertain);self.assertEqual(r['status'],'failed');self.assertIn('uncertain',r['reason']);self.assertEqual(r['exit_code'],1)
if __name__=='__main__':unittest.main()
